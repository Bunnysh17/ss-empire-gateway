import requests
import time

BASE_URL = "https://ss-empire-gateway.onrender.com"

# 1. Create Order & Generate Dynamic UPI QR
order_payload = {
    "amount": "100",               # Amount in INR
    "customer_name": "John Doe",
    "customer_mobile": "9876543210",
    "remark": "VIP Pass"
}

resp = requests.post(f"{BASE_URL}/api/create-order", json=order_payload).json()

if resp.get("success"):
    order_id = resp["order_id"]
    qr_url = resp["qr_image_url"]     # Direct PNG QR image for embeds
    upi_intent = resp.get("upi_intent")

    print(f"✅ Order Created: {order_id}")
    print(f"📸 QR Image URL: {qr_url}")
    print(f"📱 UPI Intent Deep-Link: {upi_intent}")

    # 2. Check Payment Status (Poll every 3 seconds)
    print("⏳ Awaiting customer payment...")
    for _ in range(60):  # Poll up to 3 minutes
        time.sleep(3)
        status_resp = requests.post(f"{BASE_URL}/api/check-status", json={"order_id": order_id}).json()
        
        if status_resp.get("status") == "SUCCESS":
            bank_utr = status_resp.get("utr")
            amount_paid = status_resp.get("amount")
            print(f"🎉 Payment SUCCESS! Verified ₹{amount_paid} with Bank UTR: {bank_utr}")
            break
else:
    print("❌ Failed to create order:", resp.get("message"))


            
            
import discord
from discord.ext import commands
import requests
import asyncio

bot = commands.Bot(command_prefix="!", intents=discord.Intents.all())
GATEWAY_URL = "https://ss-empire-gateway.onrender.com"

@bot.command()
async def pay(ctx, amount: str = "100"):
    """Usage: !pay 100"""
    # 1. Create Gateway Order
    res = requests.post(f"{GATEWAY_URL}/api/create-order", json={
        "amount": amount,
        "customer_name": ctx.author.name,
        "remark": f"Discord_{ctx.author.id}"
    }).json()

    if not res.get("success"):
        await ctx.send("❌ Error creating payment QR.")
        return

    order_id = res["order_id"]
    qr_url = res["qr_image_url"]

    # 2. Send Discord Embed with Live Dynamic QR
    embed = discord.Embed(
        title="⚡ SS EMPIRE — Instant UPI Payment",
        description="Scan QR with **PhonePe, Paytm, or Google Pay** (Savings Account)",
        color=0xff1744
    )
    embed.add_field(name="💰 Amount", value=f"₹{amount}", inline=True)
    embed.add_field(name="🆔 Order ID", value=f"`{order_id}`", inline=True)
    embed.set_image(url=qr_url)
    embed.set_footer(text="Awaiting payment confirmation... auto verifies!")

    msg = await ctx.send(embed=embed)

    # 3. Poll payment status asynchronously in background
    for _ in range(60): # 3 minutes timeout
        await asyncio.sleep(3)
        chk = requests.post(f"{GATEWAY_URL}/api/check-status", json={"order_id": order_id}).json()
        
        if chk.get("status") == "SUCCESS":
            success_embed = discord.Embed(
                title="✅ Payment Verified & Role Granted!",
                description=f"Thank you {ctx.author.mention}!\n**Amount:** ₹{amount}\n**Bank UTR:** `{chk.get('utr')}`",
                color=0x00e676
            )
            await msg.edit(embed=success_embed)
            # Add user VIP role here automatically:
            # role = discord.utils.get(ctx.guild.roles, name="VIP")
            # await ctx.author.add_roles(role)
            break

bot.run("YOUR_DISCORD_BOT_TOKEN")


            
            
import telebot
import requests
import time

BOT_TOKEN = "YOUR_TELEGRAM_BOT_TOKEN"
GATEWAY_URL = "https://ss-empire-gateway.onrender.com"
bot = telebot.TeleBot(BOT_TOKEN)

@bot.message_handler(commands=['pay'])
def handle_pay(message):
    try:
        # Command: /pay 100
        parts = message.text.split()
        amount = parts[1] if len(parts) > 1 else "50"

        # 1. Create Order on Gateway
        res = requests.post(f"{GATEWAY_URL}/api/create-order", json={
            "amount": amount,
            "customer_name": message.from_user.first_name or "TelegramUser",
            "remark": f"TG_{message.from_user.id}"
        }).json()

        if not res.get("success"):
            bot.reply_to(message, "❌ Error generating payment QR.")
            return

        order_id = res['order_id']
        qr_url = res['qr_image_url']

        caption = (
            f"⚡ SS EMPIRE UPI PAYMENT\n\n"
            f"💰 Amount: ₹{amount}\n"
            f"🆔 Order ID: {order_id}\n\n"
            f"📱 Scan QR using PhonePe / GPay / Paytm\n"
            f"⏳ Auto-verifying your payment in real-time..."
        )

        sent_msg = bot.send_photo(message.chat.id, qr_url, caption=caption, parse_mode="HTML")

        # 2. Poll for payment status
        for _ in range(60): # 3 minutes
            time.sleep(3)
            chk = requests.post(f"{GATEWAY_URL}/api/check-status", json={"order_id": order_id}).json()
            if chk.get('status') == 'SUCCESS':
                bot.reply_to(
                    sent_msg, 
                    f"🎉 Payment Received Successfully!\n\n"
                    f"💵 Amount: ₹{amount}\n"
                    f"🏦 Bank Reference (UTR): {chk.get('utr')}\n\n"
                    f"✅ Your access has been unlocked automatically!",
                    parse_mode="HTML"
                )
                break
    except Exception as e:
        bot.reply_to(message, f"Error: {str(e)}")

bot.infinity_polling()


            
            
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

public class SSEmpireGateway {
    private static final String BASE_URL = "https://ss-empire-gateway.onrender.com";

    public static void main(String[] args) throws Exception {
        HttpClient client = HttpClient.newHttpClient();

        // 1. Create UPI Order
        String jsonPayload = """
        {
            "amount": "100",
            "customer_name": "Rahul",
            "customer_mobile": "9876543210",
            "remark": "Java SDK Order"
        }
        """;

        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(BASE_URL + "/api/create-order"))
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(jsonPayload))
                .build();

        HttpResponse response = client.send(request, HttpResponse.BodyHandlers.ofString());
        System.out.println("✅ Order Created: " + response.body());

        // 2. Poll Status Check with Order ID
        String checkPayload = "{\"order_id\":\"TXN1789924627C92E\"}";
        HttpRequest checkReq = HttpRequest.newBuilder()
                .uri(URI.create(BASE_URL + "/api/check-status"))
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(checkPayload))
                .build();

        HttpResponse checkResp = client.send(checkReq, HttpResponse.BodyHandlers.ofString());
        System.out.println("Status Check: " + checkResp.body());
    }
}


            
            
const axios = require('axios');

const BASE_URL = 'https://ss-empire-gateway.onrender.com';

async function processPayment(amount, customerName, remark) {
  try {
    // 1. Create Order
    const { data: order } = await axios.post(`${BASE_URL}/api/create-order`, {
      amount: String(amount),
      customer_name: customerName,
      customer_mobile: '9876543210',
      remark: remark
    });

    if (!order.success) {
      console.error('Order creation failed:', order.message);
      return;
    }

    console.log(`✅ Order ID: ${order.order_id}`);
    console.log(`📸 QR Code Image: ${order.qr_image_url}`);
    console.log(`📱 UPI Intent: ${order.upi_intent}`);

    // 2. Poll Payment Status every 3s
    const pollInterval = setInterval(async () => {
      const { data: status } = await axios.post(`${BASE_URL}/api/check-status`, {
        order_id: order.order_id
      });

      if (status.status === 'SUCCESS') {
        clearInterval(pollInterval);
        console.log(`🎉 Payment Verified! UTR: ${status.utr}, Amount: ₹${status.amount}`);
      }
    }, 3000);

  } catch (error) {
    console.error('API Error:', error.message);
  }
}

processPayment(100, 'Suyash', 'VIP Pass');
